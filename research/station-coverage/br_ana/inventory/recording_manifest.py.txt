"""Retained recording files → content identities and exact data-request coverage."""
import hashlib
import json
from pathlib import Path
from rivretrieve._internal.recordings import read_recording
out=Path.cwd()/".worktrees/brazil-live-evidence"
records=[]
for path in sorted(out.glob("*.recording.json")):
    rec=read_recording(path)
    records.append({"file":path.name,"file_sha256":hashlib.sha256(path.read_bytes()).hexdigest(),"payload_sha256":rec.sha256,"payload_bytes":len(rec.content),"http_status":rec.status_code,"retrieved_at":rec.retrieved_at.isoformat(),"method":rec.request.method.value,"url":rec.request.url,"parameters":dict(rec.request.parameters or {}),"credential_header_names":list(rec.request.credential_header_names)})
(out/"recording-manifest.json").write_text(json.dumps(records,ensure_ascii=False,indent=2)+"\n")
print("manifest recordings:",len(records))
