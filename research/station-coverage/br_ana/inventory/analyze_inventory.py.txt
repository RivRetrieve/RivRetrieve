"""ANA inventory accounting : exact retained UF recordings → population and conflict report."""
import collections
import hashlib
import json
from pathlib import Path
from rivretrieve._internal.recordings import read_recording

out = Path.cwd()/".worktrees/brazil-live-evidence"
by_id = collections.defaultdict(list)
units = []
attempts = []
for result_path in sorted(out.glob("inventory-ufs*.results.json")):
    attempts.extend({"attempt_file":result_path.name,**item} for item in json.loads(result_path.read_text()))
fields = collections.Counter()
kinds = collections.Counter()
coordinate_states = collections.Counter()
flag_values = collections.defaultdict(collections.Counter)
missing_ids = []
for uf in "AC AL AM AP BA CE DF ES GO MA MT MS MG PA PB PR PE PI RJ RN RS RO RR SC SP SE TO".split():
    path = out/("inventory_UF_"+uf+".recording.json")
    if not path.exists():
        units.append({"uf":uf,"state":"no_recording"})
        continue
    rec = read_recording(path)
    envelope = json.loads(rec.content)
    rows = envelope["items"]
    units.append({"uf":uf,"http_status":rec.status_code,"source_code":envelope["code"],"rows":len(rows),"sha256":rec.sha256,"bytes":len(rec.content),"retrieved_at":rec.retrieved_at.isoformat()})
    for index,row in enumerate(rows):
        identity = row.get("codigoestacao")
        if identity is None or identity == "":
            missing_ids.append({"uf":uf,"row_index":index})
        else:
            digest=hashlib.sha256(json.dumps(row,sort_keys=True,ensure_ascii=False,separators=(",",":")).encode()).hexdigest()
            by_id[identity].append({"uf":uf,"row_index":index,"row_sha256":digest,"native_uf":row.get("UF_Estacao")})
        fields.update(row.keys())
        kinds[row.get("Tipo_Estacao")] += 1
        for name in ("Latitude","Longitude"):
            value=row.get(name)
            if value is None or value=="": state="blank_or_null"
            else:
                try: float(value); state="numeric"
                except (ValueError,TypeError): state="invalid_numeric"
            coordinate_states[name+":"+state]+=1
        for name,value in row.items():
            if name.startswith("Tipo_Estacao_"):flag_values[name][value]+=1
report={"attempts":attempts,"units":units,"total_rows":sum(x.get("rows",0) for x in units),"distinct_station_ids":len(by_id),"missing_ids":missing_ids,"duplicate_station_ids":{k:v for k,v in by_id.items() if len(v)>1},"conflicting_station_ids":[k for k,v in by_id.items() if len({x["row_sha256"] for x in v})>1],"uf_mismatch":[{"station_id":k,**x} for k,v in by_id.items() for x in v if x["uf"]!=x["native_uf"]],"fields":dict(fields),"station_types":dict(kinds),"coordinate_states":dict(coordinate_states),"type_flag_values":{k:dict(v) for k,v in flag_values.items()}}
(out/"inventory-accounting.json").write_text(json.dumps(report,ensure_ascii=False,indent=2)+"\n")
print(json.dumps({k:report[k] for k in ["total_rows","distinct_station_ids","missing_ids","conflicting_station_ids","station_types","coordinate_states"]},ensure_ascii=False))
print("duplicate station ids:",len(report["duplicate_station_ids"]),"UF mismatches:",len(report["uf_mismatch"]))
